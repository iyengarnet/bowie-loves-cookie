function moveDog(dog) { const directions = [ [1, 0], [-1, 0], [0, 1], [0, -1] ]; /* Keep moving straight whenever possible. */ if (dog.direction) { const [dx, dy] = dog.direction; const nextX = dog.x + dx; const nextY = dog.y + dy; if ( nextX >= 0 && nextX < WIDTH && nextY >= 0 && nextY < HEIGHT && !walls.has(positionKey(nextX, nextY)) ) { dog.x = nextX; dog.y = nextY; return; } } /* We reached a wall, edge, or junction. Look for available directions. */ const available = directions.filter(([dx, dy]) => { const nx = dog.x + dx; const ny = dog.y + dy; return ( nx >= 0 && nx < WIDTH && ny >= 0 && ny < HEIGHT && !walls.has(positionKey(nx, ny)) ); }); if (available.length === 0) { return; } /* Score each direction by how close it ultimately gets the dog to the Cookie. We use BFS distance rather than just straight-line distance, so dogs don't repeatedly choose a direction that looks close but leads into a dead end. */ function distanceToCookie(startX, startY) { const queue = [ [startX, startY, 0] ]; const visited = new Set([ positionKey(startX, startY) ]); while (queue.length > 0) { const [x, y, distance] = queue.shift(); if ( x === cookie.x && y === cookie.y ) { return distance; } for (const [nx, ny] of neighbors(x, y)) { const key = positionKey(nx, ny); if ( !walls.has(key) && !visited.has(key) ) { visited.add(key); queue.push([ nx, ny, distance + 1 ]); } } } return Infinity; } /* Give every available direction a distance score. */ const scored = available.map(direction => { const [dx, dy] = direction; const nx = dog.x + dx; const ny = dog.y + dy; return { direction, distance: distanceToCookie(nx, ny) }; }); /* Find the shortest route to the Cookie. */ const bestDistance = Math.min( ...scored.map(item => item.distance) ); let bestChoices = scored.filter( item => item.distance === bestDistance ); /* Don't immediately reverse direction unless it is the only sensible route. */ if (dog.direction) { const [oldDx, oldDy] = dog.direction; const nonReverse = bestChoices.filter( item => { const [newDx, newDy] = item.direction; return !( newDx === -oldDx && newDy === -oldDy ); } ); if (nonReverse.length > 0) { bestChoices = nonReverse; } } /* Pick randomly only between equally good routes, so different dogs can still take slightly different paths. */ const chosen = bestChoices[ Math.floor( Math.random() * bestChoices.length ) ]; dog.direction = chosen.direction; /* Move immediately in the new direction. */ dog.x += dog.direction[0]; dog.y += dog.direction[1]; }
